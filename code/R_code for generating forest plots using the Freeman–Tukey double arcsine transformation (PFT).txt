# ============================================================
# SINGLE-PROPORTION META-ANALYSIS TEMPLATE
# ============================================================
# This script performs single-proportion meta-analysis using the
# Freeman-Tukey double arcsine transformation (PFT) with a
# random-effects model.
#
# Outputs include:
#   - Pooled proportion with 95% CI and 95% PI
#   - Forest plots with prediction intervals
#   - Excel summary with all results
# ============================================================

library(meta)
library(readxl)
library(writexl)
library(openxlsx)
library(dplyr)

# ============================================================
# 1. FILE PATH
# ============================================================
file_path <- "D:/Desktop/meta/E.coli.xlsx"

sheet_names <- excel_sheets(file_path)
cat("Sheets found:", paste(sheet_names, collapse = ", "), "\n\n")

all_results <- list()

# ============================================================
# 2. HELPER FUNCTIONS
# ============================================================
shorten_sheet_name <- function(name, max_length = 31) {
  if (nchar(name) <= max_length) {
    return(name)
  } else {
    short_name <- substr(name, 1, 25)
    return(paste0(short_name, "..."))
  }
}

# Back-transform PFT summary values to the original proportion scale
backtransform_pft <- function(x, n_vector) {
  if (length(x) == 0 || is.null(x) || all(is.na(x))) return(NA_real_)
  n_vector <- n_vector[is.finite(n_vector) & n_vector > 0]
  if (length(n_vector) == 0) return(NA_real_)

  harmonic_n <- 1 / mean(1 / n_vector)

  as.numeric(
    meta::backtransf(
      x,
      sm = "PFT",
      n = harmonic_n
    )
  )
}

# ============================================================
# 3. LOOP THROUGH EACH SHEET
# ============================================================
for (sheet_name in sheet_names) {

  cat("Analyzing:", sheet_name, "\n")
  ecoli <- read_excel(file_path, sheet = sheet_name)

  if (!all(c("Events", "Total", "study") %in% names(ecoli))) {
    cat("WARNING:", sheet_name, "missing required columns - skipping\n\n")
    next
  }

  ecoli <- ecoli %>%
    mutate(
      Events = suppressWarnings(as.numeric(Events)),
      Total = suppressWarnings(as.numeric(Total))
    )

  ecoli_clean <- ecoli %>%
    filter(
      !is.na(Events),
      !is.na(Total),
      Total >= 10,
      Events >= 0,
      Events <= Total
    )

  if (nrow(ecoli_clean) == 0) {
    cat("WARNING:", sheet_name, "no valid data after cleaning - skipping\n\n")
    next
  }

  complete_data <- ecoli %>%
    filter(!is.na(Events), !is.na(Total))

  removed_na <- nrow(ecoli) - nrow(complete_data)

  small_studies <- complete_data %>% filter(Total < 10)
  removed_small <- nrow(small_studies)

  invalid_studies <- complete_data %>%
    filter(Total <= 0 | Events < 0 | Events > Total)
  removed_invalid <- nrow(invalid_studies)

  total_removed <- nrow(ecoli) - nrow(ecoli_clean)

  cat("  Data cleaning summary:\n")
  cat("  - Original studies:", nrow(ecoli), "\n")
  if (removed_na > 0) cat("  - Removed due to NA values:", removed_na, "\n")
  if (removed_small > 0) {
    cat("  - Removed due to Total < 10:", removed_small, "\n")
    cat("  - Removed small-sample studies:", paste(small_studies$study, collapse = ", "), "\n")
  }
  if (removed_invalid > 0) cat("  - Removed due to invalid Events/Total:", removed_invalid, "\n")
  cat("  - Final studies included:", nrow(ecoli_clean), "\n")

  cat("  Data type check:\n")
  cat("  - Events type:", class(ecoli_clean$Events), "\n")
  cat("  - Total type:", class(ecoli_clean$Total), "\n")
  cat("  - Events range:", min(ecoli_clean$Events), "to", max(ecoli_clean$Events), "\n")
  cat("  - Total range:", min(ecoli_clean$Total), "to", max(ecoli_clean$Total), "\n")

  # ==========================================================
  # 4. MAIN META-ANALYSIS: PFT + Random effects + REML + 95% PI
  # ==========================================================
  tryCatch({

    meta.overall <- metaprop(
      event = Events,
      n = Total,
      data = ecoli_clean,
      studlab = study,
      sm = "PFT",
      method.tau = "REML",
      common = FALSE,
      random = TRUE,
      overall = TRUE,
      prediction = TRUE,
      level.predict = 0.95,
      backtransf = TRUE
    )

    pooled_prop <- backtransform_pft(meta.overall$TE.random, meta.overall$n)
    pooled_lower <- backtransform_pft(meta.overall$lower.random, meta.overall$n)
    pooled_upper <- backtransform_pft(meta.overall$upper.random, meta.overall$n)

    pi_lower <- backtransform_pft(meta.overall$lower.predict, meta.overall$n)
    pi_upper <- backtransform_pft(meta.overall$upper.predict, meta.overall$n)

    # ========================================================
    # 5. FOREST PLOT WITH PREDICTION INTERVAL
    # ========================================================
    pdf_file <- paste0("D:/Desktop/meta/", sheet_name, "_forest.pdf")
    pdf_height <- max(20, 0.35 * meta.overall$k + 8)

    pdf(pdf_file, height = pdf_height, width = 14, family = "Times")

    forest(
      meta.overall,
      layout = "Revman",
      leftcols = c("studlab", "event", "n", "effect.ci"),
      leftlabs = c("Study", "Resistant", "Total", "Proportion [95% CI]"),
      rightcols = FALSE,
      family = "Times",
      fontsize = 9.5,

      col.vertical.line = "transparent",
      col.diamond.common = "lightslategray",
      col.diamond.lines.common = "lightslategray",
      col.diamond.random = "#669999",
      col.diamond.lines.random = "#669999",
      col.square = "#336699",
      col.square.lines = "#336699",
      col.study = "#006666",
      col.inside = "black",
      col.label.right = "black",
      col.random = "#660099",
      col.predict = "#660099",

      overall = TRUE,
      overall.hetstat = TRUE,
      common = FALSE,
      random = TRUE,
      prediction = TRUE,
      backtransf = TRUE,

      weight.study = "same",
      squaresize = 0.8,
      height = 0.5,

      plotwidth = "12cm",
      colgap.forest.left = "1.5cm",
      colgap.left = "1.5cm",

      xlab = "",
      xlim = c(0, 1),
      smlab = "Proportion",

      # Number and p-value display:
      # Heterogeneity p-values < 0.001 are displayed as P < 0.001
      digits = 2,
      digits.pval = 3,
      digits.pval.Q = 3,
      scientific.pval = FALSE,
      JAMA.pval = TRUE,
      zero.pval = TRUE
    )

    dev.off()
    cat("Forest plot saved:", pdf_file, "\n")

    # ========================================================
    # 6. SUMMARY RESULTS WITH PI AND CORRECT PFT BACK-TRANSFORMATION
    # ========================================================
    results_df <- data.frame(
      Sheet_Name = sheet_name,
      Studies_Original = nrow(ecoli),
      Studies_Analyzed = meta.overall$k,
      Studies_Removed_NA = removed_na,
      Studies_Removed_Small = removed_small,
      Studies_Removed_Invalid = removed_invalid,
      Total_Removed = total_removed,
      Total_Participants = sum(meta.overall$n),
      Total_Events = sum(meta.overall$event),

      Overall_Proportion = round(pooled_prop, 4),
      Overall_Percent = round(pooled_prop * 100, 2),

      CI_95_Lower = round(pooled_lower, 4),
      CI_95_Upper = round(pooled_upper, 4),
      CI_95_Lower_Percent = round(pooled_lower * 100, 2),
      CI_95_Upper_Percent = round(pooled_upper * 100, 2),

      PI_95_Lower = round(pi_lower, 4),
      PI_95_Upper = round(pi_upper, 4),
      PI_95_Lower_Percent = round(pi_lower * 100, 2),
      PI_95_Upper_Percent = round(pi_upper * 100, 2),

      Tau2 = round(meta.overall$tau2, 4),
      I2 = round(meta.overall$I2 * 100, 2),
      P_value_heterogeneity = format.pval(meta.overall$pval.Q, eps = 0.001, digits = 3),
      Q_statistic = round(meta.overall$Q, 4),
      df = meta.overall$df.Q,
      Status = "Success"
    )

    all_results[[sheet_name]] <- list(
      summary = results_df,
      meta_object = meta.overall,
      cleaned_data = ecoli_clean
    )

    cat("Analysis results:", meta.overall$k, "studies\n")
    cat("  Pooled proportion = ", round(pooled_prop, 4), " (", round(pooled_prop * 100, 2), "%)\n", sep = "")
    cat("  95% CI = ", round(pooled_lower, 4), " - ", round(pooled_upper, 4), "\n", sep = "")
    cat("  95% Prediction Interval = ", round(pi_lower, 4), " - ", round(pi_upper, 4), "\n", sep = "")
    cat("  Tau² = ", round(meta.overall$tau2, 4), "; I² = ", round(meta.overall$I2 * 100, 2), "%\n\n", sep = "")

  }, error = function(e) {

    if (dev.cur() > 1) try(dev.off(), silent = TRUE)
    cat("  Meta-analysis error:", e$message, "\n")

    results_df <- data.frame(
      Sheet_Name = sheet_name,
      Studies_Original = nrow(ecoli),
      Studies_Analyzed = nrow(ecoli_clean),
      Studies_Removed_NA = removed_na,
      Studies_Removed_Small = removed_small,
      Studies_Removed_Invalid = removed_invalid,
      Total_Removed = total_removed,
      Total_Participants = sum(ecoli_clean$Total, na.rm = TRUE),
      Total_Events = sum(ecoli_clean$Events, na.rm = TRUE),
      Overall_Proportion = NA,
      Overall_Percent = NA,
      CI_95_Lower = NA,
      CI_95_Upper = NA,
      CI_95_Lower_Percent = NA,
      CI_95_Upper_Percent = NA,
      PI_95_Lower = NA,
      PI_95_Upper = NA,
      PI_95_Lower_Percent = NA,
      PI_95_Upper_Percent = NA,
      Tau2 = NA,
      I2 = NA,
      P_value_heterogeneity = NA,
      Q_statistic = NA,
      df = NA,
      Status = paste("Error:", e$message)
    )

    all_results[[sheet_name]] <- list(
      summary = results_df,
      meta_object = NULL,
      cleaned_data = ecoli_clean
    )

    cat("  Error recorded, continuing to next sheet\n\n")
  })
}

# ============================================================
# 7. OUTPUT EXCEL
# ============================================================
if (length(all_results) == 0) {

  cat("WARNING: No sheets contained valid data for analysis!\n")

} else {

  all_summaries <- do.call(rbind, lapply(all_results, function(x) x$summary))

  results_file <- "D:/Desktop/meta/All_Sheets_Meta_Analysis_Results_with_PI.xlsx"
  wb <- createWorkbook()

  addWorksheet(wb, "Summary_All_Sheets")
  writeData(wb, "Summary_All_Sheets", all_summaries)

  used_sheet_names <- "Summary_All_Sheets"

  for (sheet_name in names(all_results)) {

    short_name <- shorten_sheet_name(sheet_name)

    if (!is.null(all_results[[sheet_name]]$meta_object)) {

      m <- all_results[[sheet_name]]$meta_object

      # Study-level results: output raw Events/Total proportions
      study_details <- data.frame(
        Study = m$studlab,
        Events = m$event,
        Total = m$n,
        Raw_Proportion = round(m$event / m$n, 4),
        Raw_Percent = round((m$event / m$n) * 100, 2),
        Weight_Random = round(m$w.random, 2)
      )

      details_sheet_name <- substr(paste0("Det_", short_name), 1, 31)

      if (details_sheet_name %in% used_sheet_names) {
        details_sheet_name <- substr(paste0(details_sheet_name, "_2"), 1, 31)
      }

      addWorksheet(wb, details_sheet_name)
      writeData(wb, details_sheet_name, study_details)
      used_sheet_names <- c(used_sheet_names, details_sheet_name)
    }

    cleaned_sheet_name <- substr(paste0("Data_", short_name), 1, 31)

    if (cleaned_sheet_name %in% used_sheet_names) {
      cleaned_sheet_name <- substr(paste0(cleaned_sheet_name, "_2"), 1, 31)
    }

    addWorksheet(wb, cleaned_sheet_name)
    writeData(wb, cleaned_sheet_name, all_results[[sheet_name]]$cleaned_data)
    used_sheet_names <- c(used_sheet_names, cleaned_sheet_name)
  }

  # Methods / version record
  addWorksheet(wb, "Methods_Readme")

  methods_df <- data.frame(
    Item = c(
      "Outcome",
      "Summary measure",
      "Random-effects model",
      "Tau-squared estimator",
      "Prediction interval",
      "Prediction interval level",
      "PFT back-transformation",
      "Minimum Total filter",
      "R version",
      "meta package version"
    ),
    Setting = c(
      "Single proportion",
      "Freeman-Tukey double-arcsine transformation (PFT)",
      "Random effects",
      "REML",
      "Yes",
      "95%",
      "Using harmonic mean of study sample sizes",
      "Total >= 10",
      R.version.string,
      as.character(packageVersion("meta"))
    ),
    stringsAsFactors = FALSE
  )

  writeData(wb, "Methods_Readme", methods_df)

  saveWorkbook(wb, results_file, overwrite = TRUE)

  cat("\n=== Analysis complete ===\n")
  cat("All sheets processed!\n")
  cat("Forest plots saved in: D:/Desktop/meta/\n")
  cat("Summary results saved in:", results_file, "\n")
  cat("Sheets analyzed successfully:", sum(all_summaries$Status == "Success"), "\n")
  cat("Sheets with errors:", sum(all_summaries$Status != "Success"), "\n")

  cat("\n=== Data cleaning summary ===\n")
  for (sheet_name in names(all_results)) {
    result <- all_results[[sheet_name]]$summary
    cat(sheet_name, ": ", result$Studies_Original, "original studies -> ",
        result$Studies_Analyzed, "analyzed (status:", result$Status, ")\n")

    if (result$Studies_Removed_NA > 0 |
        result$Studies_Removed_Small > 0 |
        result$Studies_Removed_Invalid > 0) {
      cat("     Removed: NA(", result$Studies_Removed_NA,
          "), Total<10(", result$Studies_Removed_Small,
          "), invalid(", result$Studies_Removed_Invalid, ")\n", sep = "")
    }
  }

  cat("\n=== 95% Prediction Interval summary ===\n")
  successful_results <- all_summaries %>%
    filter(Status == "Success") %>%
    select(
      Sheet_Name,
      Overall_Percent,
      CI_95_Lower_Percent,
      CI_95_Upper_Percent,
      PI_95_Lower_Percent,
      PI_95_Upper_Percent,
      Tau2,
      I2
    )

  print(successful_results)
}
